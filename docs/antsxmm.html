 33    Parses a BIDS directory generated by antsxbids.

 42    
 43    sessions_data = []
 44
 45    for sub_dir in subjects:
 46        sub_id = sub_dir.name
 47        
 48        sessions = sorted([d for d in sub_dir.glob("ses*") if d.is_dir()])
 49        
 50        for ses_dir in sessions:
 51            ses_id = ses_dir.name
 52            
 53            data = {
 54                'subjectID': sub_id,
 55                'date': ses_id, 
 56                'session_path': str(ses_dir)
 57            }
 58
 59            # 1. Anatomy (T1w, FLAIR)
 60            anat_dir = ses_dir / 'anat'
 61            t1_files = []
 62            if anat_dir.exists():
 63                t1_files = _sorted_glob(anat_dir, ["*T1w.nii.gz", "*T1w.nii"])
 64                flair_files = _sorted_glob(anat_dir, ["*FLAIR.nii.gz", "*FLAIR.nii"])
 65                t2w_files = _sorted_glob(anat_dir, ["*T2w.nii.gz", "*T2w.nii"])
 66
 67                data['flair_filenames'] = flair_files
 68                data['t2w_filenames'] = t2w_files
 69
 70                # Back-compat singletons
 71                data['flair_filename'] = _select_first(flair_files)
 72                data['t2w_filename'] = _select_first(t2w_files)
 73            
 74            data['t1_filenames'] = t1_files # Store ALL T1s
 75
 76            # 2. DWI
 77            dwi_dir = ses_dir / 'dwi'
 78            dwi_files = []
 79            if dwi_dir.exists():
 80                dwi_files = _sorted_glob(dwi_dir, ["*.nii.gz", "*.nii"])
 81            data['dti_filenames'] = dwi_files
 82
 83            # 3. Functional
 84            func_dir = ses_dir / 'func'
 85            rsf_files = []
 86            if func_dir.exists():
 87                rsf_files = _sorted_glob(func_dir, ["*bold.nii.gz", "*bold.nii"])
 88            data['rsf_filenames'] = rsf_files
 89
 90            # 4. Neuromelanin
 91            nm_files = []
 92            for nm_dir in (ses_dir / 'melanin', ses_dir / 'anat'):
 93                if nm_dir.exists():
 94                    nm_files.extend(_sorted_glob(nm_dir, ["*NM.nii.gz", "*NM.nii"]))
 95            data['nm_filenames'] = sorted(set(nm_files))
 96
 97            # 5. Perfusion
 98            perf_dir = ses_dir / 'perf'
 99            if perf_dir.exists():
100                perf_files = _sorted_glob(perf_dir, ["*.nii.gz", "*.nii"])
101                data['perf_filenames'] = perf_files
102                # Prefer ASL-like series if present.
103                data['perf_filename'] = _select_first(perf_files, preferred_substrings=["asl", "cbf", "perfusion"]) 
104
105            # 6. PET
106            pet_dir = ses_dir / 'pet'
107            if pet_dir.exists():
108                pet_files = _sorted_glob(pet_dir, ["*.nii.gz", "*.nii"])
109                data['pet3d_filenames'] = pet_files
110                # Prefer typical PET series names if present.
111                data['pet3d_filename'] = _select_first(pet_files, preferred_substrings=["pet", "suv", "suvr", "dyn", "static"]) 
112
113            # Only add if we have at least one T1
114            if data['t1_filenames']:
115                sessions_data.append(data)
116
117    return pd.DataFrame(sessions_data)
