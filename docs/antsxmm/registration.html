def bvec_reorientation( motion_parameters: list[typing.Any] | None, bvecs: numpy.ndarray, rebase: numpy.ndarray | None = None) -> numpy.ndarray: 
187def bvec_reorientation(
188    motion_parameters: list[Any] | None,
189    bvecs: np.ndarray,
190    rebase: np.ndarray | None = None,
191) -> np.ndarray:
192    """Reorient b-vectors using motion correction transformation parameters."""
193    if motion_parameters is None:
194        raise ValueError("motion_parameters cannot be None")
195    bvecs = np.asarray(bvecs, dtype=float).copy()
196    for myidx in range(len(motion_parameters)):
197        if motion_parameters[myidx] != "NA":
198            rotmat = ants.read_transform(motion_parameters[myidx][0]).parameters[0:9].reshape(3, 3)
199            det = np.linalg.det(rotmat)
200            if abs(det) > 1e-8:
201                u, _, vt = np.linalg.svd(rotmat)
202                rotation_matrix = u @ vt
203                bvecs[myidx, :] = rotation_matrix @ bvecs[myidx, :]
204            norm = np.linalg.norm(bvecs[myidx, :])
205            if norm > 1e-8:
206                bvecs[myidx, :] = bvecs[myidx, :] / norm
207            if rebase is not None:
208                bvecs[myidx, :] = rebase @ bvecs[myidx, :]
209    return bvecs

def deformation_gradient_optimized( warp_image: ants.core.ants_image.ANTsImage, to_rotation: bool = False, to_inverse_rotation: bool = False) -> numpy.ndarray: 
 86def deformation_gradient_optimized(
 87    warp_image: ants.ANTsImage,
 88    to_rotation: bool = False,
 89    to_inverse_rotation: bool = False,
 90) -> np.ndarray:
 91    """Compute the deformation gradient tensor from a displacement (warp) field image."""
 92    if not ants.is_image(warp_image):
 93        raise RuntimeError("ANTsImage is required")
 94    dim = warp_image.dimension
 95    tshp = warp_image.shape
 96    tdir = warp_image.direction
 97    spc = warp_image.spacing
 98    warpnp = warp_image.numpy()
 99    gradient_list = [np.gradient(warpnp[..., k], *spc, axis=range(dim)) for k in range(dim)]
100    dg = np.stack([np.stack(grad_k, axis=-1) for grad_k in gradient_list], axis=-1)
101    dg = (tdir @ dg).swapaxes(-1, -2)
102    dg += np.eye(dim)
103    if to_rotation or to_inverse_rotation:
104        u, _, vh = np.linalg.svd(dg)
105        z = u @ vh
106        dets = np.linalg.det(z)
107        reflection_mask = dets < 0
108        vh[reflection_mask, -1, :] *= -1
109        z[reflection_mask] = u[reflection_mask] @ vh[reflection_mask]
110        dg = z
111        if to_inverse_rotation:
112            dg = np.transpose(dg, axes=(*range(dg.ndim - 2), dg.ndim - 1, dg.ndim - 2))
113    new_shape = tshp + (dim, dim)
114    return np.reshape(dg, new_shape)
